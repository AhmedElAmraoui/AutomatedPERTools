	\section{Sparse Pauli Tomography}
	
	% --- Sparse Pauli Tomography ---
	\[
	N_{\mathrm{tomo}}
	= N_{\mathrm{layers}} \cdot \bigl( 9 \cdot |D| \cdot S \bigr)
	+ \sum_{\ell=1}^{N_{\mathrm{layers}}} 
	\Bigl( \, \lvert \text{single\_bases}(\ell) \rvert \cdot S_{\mathrm{single}} \Bigr)
	\]
	
	\begin{align*}
		&N_{\mathrm{layers}} && \text{= Anzahl der Clifford-Layer im Circuit} \\
		&9 && \text{= feste Zahl der Messbasen} \\
		&|D| && \text{= Anzahl der Tiefenwerte (z.\,B. [2,4,16,32,64])} \\
		&S && \text{= Samples (Twirls pro Basis)} \\
		&|\text{single\_bases}(\ell)| && \text{= Zahl der Single-Bases für Layer $\ell$} \\
		&S_{\mathrm{single}} && \text{= Samples für die Single-Bases}
	\end{align*}
	
	\section{PER} 
	% --- PER ---
	\[
	N_{\mathrm{PER}}
	= N_{\mathrm{input}} \cdot N_{\mathrm{meas\_bases}}^{\mathrm{PER}} \cdot |N_{\mathrm{strengths}}| \cdot S
	\]
	
	\begin{align*}
		&N_{\mathrm{input}} && \text{= Anzahl der Eingabecircuits} \\
		&N_{\mathrm{meas\_bases}}^{\mathrm{PER}} && \text{= minimale Zahl Messbasen, die \texttt{pauli\_list} abdeckt} \\
		&|N_{\mathrm{strengths}}| && \text{= Anzahl der verschiedenen Noise-Stärken} \\
		&S && \text{= Samples (Twirls pro Basis)}
	\end{align*}
